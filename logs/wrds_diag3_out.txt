@echo off
REM Job-queue watcher: executes any .bat dropped into GBC_data\queue\, moves it
REM to queue\done\ with a log. Writes a heartbeat each loop (used for self-heal).
setlocal enabledelayedexpansion
set Q=%USERPROFILE%\Desktop\GBC_data\queue
mkdir "%Q%" 2>nul
mkdir "%Q%\done" 2>nul
echo Queue watcher running. Drop .bat files into %Q%
:loop
echo %date% %time% > "%USERPROFILE%\Desktop\GBC_data\watcher_heartbeat.txt"
for %%f in ("%Q%\*.bat") do (
  echo === running %%~nxf at !date! !time! === >> "%Q%\done\%%~nxf.log"
  call "%%f" < nul >> "%Q%\done\%%~nxf.log" 2>&1
  move /y "%%f" "%Q%\done\" >nul
  echo === finished %%~nxf === >> "%Q%\done\%%~nxf.log"
)
timeout /t 10 /nobreak >nul
goto loop

"""WRDS pulls for the P-vs-Q project. REQUIRES one-time setup by Sean:
    pip install wrds && python -c "import wrds; wrds.Connection()"   (creates .pgpass)
Pulls: (1) OptionMetrics option bid/ask by name/date (30d ATM-to-10%ile band)
       (2) CRSP delisting returns
       (3) OptionMetrics volatility surface incl. SHORT tenors (weeklies era)
Writes csv.gz files into data/wrds/. Chunked by year to respect query limits.
"""
import os, sys
import pandas as pd
try:
    import wrds
except ImportError:
    sys.exit("pip install wrds first")
OUT = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'data', 'wrds')
os.makedirs(OUT, exist_ok=True)
db = wrds.Connection(wrds_username='seanqin2028')  # password from pgpass only

# universe: secids for our tickers
tks = [l.strip() for l in open(os.path.join(os.path.dirname(__file__), 'wrds_tickers.txt'))]
tkl = ",".join(f"'{t}'" for t in tks)
sec = db.raw_sql(f"select secid, ticker, cusip from optionm.securd where ticker in ({tkl})")
sec.to_csv(os.path.join(OUT, 'secids.csv'), index=False)
secl = ",".join(str(int(s)) for s in sec.secid.unique())

# (1)+(3) surface & quotes: opprcd has per-option best_bid/best_offer; use
# std monthly file for spreads at the strikes we trade: pull per year
for yr in range(2005, 2027):
    f1 = os.path.join(OUT, f'spreads_{yr}.csv.gz')
    if os.path.exists(f1):
        continue
    try:
        q = f"""
        select o.secid, o.date, o.exdate, o.cp_flag, o.strike_price, o.best_bid,
               o.best_offer, o.impl_volatility, o.volume, o.open_interest, o.delta
        from optionm.opprcd{yr} o
        where o.secid in ({secl}) and o.cp_flag='P'
          and o.exdate - o.date between 20 and 40
          and o.delta between -0.45 and -0.03
        """
        d = db.raw_sql(q)
        d.to_csv(f1, index=False, compression='gzip')
        print(yr, len(d), flush=True)
    except Exception as e:
        print(yr, 'err', str(e)[:120], flush=True)

# (2) CRSP delisting returns
try:
    d = db.raw_sql("""
      select permno, dlstdt, dlstcd, dlret, dlretx from crsp.dsedelist
      where dlstdt >= '1996-01-01'
    """)
    d.to_csv(os.path.join(OUT, 'delist_returns.csv.gz'), index=False, compression='gzip')
    print('delist', len(d))
except Exception as e:
    print('delist err', str(e)[:120])

# (3) short-tenor surface (weeklies): 10-day standardized surface
for yr in range(2011, 2027):
    f3 = os.path.join(OUT, f'surf10d_{yr}.csv.gz')
    if os.path.exists(f3):
        continue
    try:
        d = db.raw_sql(f"""
          select secid, date, days, delta, impl_volatility
          from optionm.stdopd{yr}
          where secid in ({secl}) and days in (10, 30) and cp_flag='P'
        """)
        d.to_csv(f3, index=False, compression='gzip')
        print('surf', yr, len(d), flush=True)
    except Exception as e:
        print('surf', yr, 'err', str(e)[:120], flush=True)
db.close()
print('WRDS PULLS DONE')
